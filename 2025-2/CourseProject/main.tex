\section{Visão geral}
O \textit{Continual Learning} (ou aprendizado contínuo) é um paradigma de aprendizado de máquina que busca desenvolver modelos capazes de aprender de forma sequencial, incorporando novos conhecimentos sem esquecer o que foi previamente aprendido. Diferentemente do aprendizado tradicional, em que o modelo é treinado uma única vez sobre um conjunto fixo de dados, o aprendizado contínuo enfrenta o desafio do \textit{catastrophic forgetting}, isto é, a perda drástica de desempenho em tarefas anteriores ao aprender novas. Para mitigar esse problema, diversas abordagens têm sido propostas, incluindo regularização dos parâmetros, uso de memória exemplar e arquiteturas dinâmicas que se adaptam ao longo do tempo.

Ao treinar sequencialmente em um tarefa A e depois em B, modelos tendem a esquecer A. Estudaremos o \emph{trade-off} estabilidade–plasticidade em duas etapas complementares:
\begin{itemize}
  \item \textbf{Parte 1} (minimização): penalização fixa de deriva para controlar o esquecimento.
  \item \textbf{Parte 2} (baselines avançados): comparação do método de penalização fi
  \item \textbf{Parte 3} (min–max): penalização adaptativa via jogo entre parâmetros (abordagem min-max) e um multiplicador~$\lambda$.
\end{itemize}

\subsection{Dados e modelo}

\textbf{Split--MNIST}: neste experimento, o conjunto MNIST é dividido em duas tarefas disjuntas: 
A = \{0–4\} e B = \{5–9\}. Cada tarefa é tratada como um problema de classificação independente, 
permitindo avaliar a capacidade do modelo em reter o conhecimento adquirido na tarefa A 
após o treinamento sequencial na tarefa B.  

O modelo pode ser um \textit{Multi-Layer Perceptron} (MLP) simples, com uma ou duas camadas ocultas e ativação ReLU, 
ou uma pequena \textit{Convolutional Neural Network} (CNN) adequada à natureza visual do MNIST. 
A função de perda utilizada é a \textit{cross-entropy}, 
e a otimização é realizada via \textit{stochastic gradient descent} (SGD) ou Adam.  
Durante o treinamento sequencial, são registradas métricas de desempenho nas duas tarefas 
para avaliar o grau de esquecimento (\textit{forgetting}) e a capacidade de transferência (\textit{transfer}).

\subsection*{Notação}

Seja $\bm{\theta}$ o vetor de parâmetros atuais do modelo e $\bm{\theta}_A$ 
a solução obtida após o treinamento completo na tarefa A.  
A quantidade abaixo mede o quanto os parâmetros se afastaram da solução anterior, 
caracterizando a \textbf{deriva dos pesos} ao longo do aprendizado contínuo:
\[
F_{\text{drift}}(\bm{\theta};\bm{\theta}_A)
= \tfrac{1}{2}\,\|\bm{\theta}-\bm{\theta}_A\|_2^2.
\]
Essa medida serve tanto como diagnóstico quanto como termo de regularização 
em abordagens que buscam equilibrar \textit{plasticidade} (adaptação à nova tarefa) 
e \textit{estabilidade} (retenção do conhecimento anterior).

\section{Desenvolvimento}

\section*{Parte 1 — Trade-off com penalização fixa (sem min–max)}
\textbf{Objetivo.} Quantificar e visualizar o conflito entre aprender B e reter A, variando a intensidade da penalização.

\subsection*{Formulação de otimização}
Após treinar A e salvar $\bm{\theta}_A$, otimize na Tarefa B:
\[
\min_{\bm{\theta}}\; \mathcal{L}_B(\bm{\theta}) \;+\; \lambda\,F_{\text{drift}}(\bm{\theta};\bm{\theta}_A),
\]
com grade inicial $\lambda \in \{0,10^{-4},10^{-3},10^{-2},10^{-1}\}$.

\subsection*{Procedimento}
\begin{enumerate}
  \item \textbf{Treino em A} até um número fixo de épocas ou acurácia-alvo; salve $\bm{\theta}_A$ e separe um conjunto de validação $\mathcal{V}_A$.
  \item \textbf{Treino em B} para cada $\lambda$, usando o mesmo otimizador e esquema de treino.
  \item \textbf{Rastreio por iteração/época}:
    \begin{itemize}
      \item Acurácia em A e em B.
      \item Deriva $\|\bm{\theta}_t-\bm{\theta}_A\|_2$.
      \item Esquecimento em A: $\Delta\mathrm{acc}_A = \mathrm{acc}_A^{\text{após A}} - \mathrm{acc}_A^{\text{após B}}$.
    \end{itemize}
\end{enumerate}

\subsection*{Plots obrigatórios}
\begin{itemize}
  \item Curvas \textbf{acc(A)} e \textbf{acc(B)} vs. iteração.
  \item \textbf{Deriva} $\|\bm{\theta}_t-\bm{\theta}_A\|_2$ vs. iteração.
  \item \textbf{Trade-off}: pontos finais de cada $\lambda$ no plano Esquecimento (x) vs. acc(B) (y).
\end{itemize}

\section*{Baseline avançado obrigatório (escolher um)}
\paragraph{EWC.}
Importâncias por parâmetro após A:
\[
F_i \approx \frac{1}{|\mathcal{D}_A|}\sum_{(x,y)\in\mathcal{D}_A}
\Big(\frac{\partial}{\partial \theta_i}\,\ell(f_{\theta}(x),y)\Big)^2\Big|_{\theta=\theta_A},
\quad
\mathcal{L}_B(\bm{\theta}) + \lambda \sum_i F_i\,(\theta_i-\theta_{A,i})^2.
\]

\paragraph{Synaptic Intelligence (SI).}
Durante A, acumule $\omega_i \leftarrow \omega_i + \Delta\theta_i \cdot g_i$. Ao final,
$\Omega_i=\frac{\omega_i}{(\Delta\theta_i)^2+\xi}$, $\xi\approx 10^{-3}$, e use
$\mathcal{L}_B(\bm{\theta}) + \lambda \sum_i \Omega_i\,(\theta_i-\theta_{A,i})^2$.

\section*{Parte 2 — Min–max com passos alternados e estabilização}
\textbf{Objetivo.} Substituir o $\lambda$ fixo por um jogo adaptativo:
\[
\min_{\bm{\theta}}\;\max_{\lambda\ge 0}\;\; \mathcal{L}_B(\bm{\theta}) - \lambda\,F(\bm{\theta};\bm{\theta}_A),
\]
com $F = F_{\text{drift}}$ para começar, passos alternados em $\bm{\theta}$ (descida) e $\lambda$ (subida), e projeção $\lambda \leftarrow \max(0,\lambda)$ a cada atualização. Estabilize com \textbf{OGDA} ou \textbf{Extragradient}.

\subsection*{Atualizações obrigatórias}
\paragraph{Alternating steps.}
A cada iteração: (i) desça em $\bm{\theta}$; (ii) suba em $\lambda$; (iii) projete $\lambda$ no semieixo positivo.

\paragraph{OGDA.}
Com $f(\bm{\theta},\lambda)=\mathcal{L}_B(\bm{\theta})-\lambda F(\bm{\theta};\bm{\theta}_A)$:
\[
\bm{\theta}_{t+1} = \bm{\theta}_{t} - \eta_\theta\big(2\nabla_{\bm{\theta}} f_t - \nabla_{\bm{\theta}} f_{t-1}\big),\quad
\lambda_{t+1} = \big[\lambda_{t} + \eta_\lambda\big(2\nabla_{\lambda} f_t - \nabla_{\lambda} f_{t-1}\big)\big]_+.
\]

\paragraph{Extragradient.}
Dois estágios:
\[
(\hat{\bm{\theta}},\hat{\lambda}) = (\bm{\theta}_{t},\lambda_t) - (\eta_\theta,-\eta_\lambda)\odot \nabla f(\bm{\theta}_t,\lambda_t),
\]
\[
(\bm{\theta}_{t+1},\lambda_{t+1}) = (\bm{\theta}_{t},\lambda_t) - (\eta_\theta,-\eta_\lambda)\odot \nabla f(\hat{\bm{\theta}},\hat{\lambda}),\quad
\lambda_{t+1}\leftarrow[\lambda_{t+1}]_+.
\]

\subsection*{Métricas e comparação}
Além das métricas da Parte 1, registre a trajetória de $\lambda$. Compare OGDA e Extragradient em estabilidade das curvas, tempo até entrar em regime estável e desempenho final.

# simple_drift_example.py
# Minimal example: train on Task A, then fine-tune on Task B with fixed lambda
# Objective:  L_B(theta) + lambda * 0.5 * ||theta - theta_A||^2

import argparse, random
import numpy as np
import torch, torch.nn as nn, torch.nn.functional as F
import torch.optim as optim
from torch.utils.data import DataLoader, Subset
from torchvision import datasets, transforms

def set_seed(seed=42):
    random.seed(seed); np.random.seed(seed)
    torch.manual_seed(seed); torch.cuda.manual_seed_all(seed)
    torch.backends.cudnn.deterministic = True
    torch.backends.cudnn.benchmark = False

class SplitMNIST(torch.utils.data.Dataset):
    def __init__(self, root, train, digits, transform, download=True):
        self.base = datasets.MNIST(root=root, train=train,
                                   transform=transform, download=download)
        self.idx = [i for i, (_, y) in enumerate(self.base) if y in digits]
    def __len__(self): return len(self.idx)
    def __getitem__(self, i): return self.base[self.idx[i]]

class MLP(nn.Module):
    def __init__(self, hidden=256, num_classes=10):
        super().__init__()
        self.fc1 = nn.Linear(28*28, hidden)
        self.fc2 = nn.Linear(hidden, num_classes)
    def forward(self, x):
        x = x.view(x.size(0), -1)
        x = F.relu(self.fc1(x))
        return self.fc2(x)

@torch.no_grad()
def evaluate(model, loader, device):
    model.eval(); ce = nn.CrossEntropyLoss()
    total, correct, loss_sum = 0, 0, 0.0
    for x, y in loader:
        x, y = x.to(device), y.to(device)
        logits = model(x); loss = ce(logits, y)
        loss_sum += loss.item() * x.size(0)
        pred = logits.argmax(1); correct += (pred == y).sum().item()
        total += x.size(0)
    return loss_sum/total, correct/total

def flatten_params(model):
    return torch.cat([p.detach().view(-1) for p in model.parameters()])

def train_task_A(model, loader, device, epochs=3, lr=1e-3):
    ce = nn.CrossEntropyLoss()
    opt = optim.Adam(model.parameters(), lr=lr)
    for _ in range(epochs):
        model.train()
        for x, y in loader:
            x, y = x.to(device), y.to(device)
            opt.zero_grad()
            loss = ce(model(x), y)
            loss.backward()
            opt.step()
    theta_A = flatten_params(model).clone().detach()
    return theta_A

def finetune_task_B_with_drift(model, loaderB, device, theta_A, lam=1e-3, epochs=3, lr=1e-3):
    ce = nn.CrossEntropyLoss()
    opt = optim.Adam(model.parameters(), lr=lr)
    for _ in range(epochs):
        model.train()
        for x, y in loaderB:
            x, y = x.to(device), y.to(device)
            opt.zero_grad()
            logits = model(x); L_B = ce(logits, y)
            theta = flatten_params(model)
            F_drift = 0.5 * torch.sum((theta - theta_A)**2)
            obj = L_B + lam * F_drift
            obj.backward()
            opt.step()

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--seed', type=int, default=42)
    ap.add_argument('--epochsA', type=int, default=3)
    ap.add_argument('--epochsB', type=int, default=3)
    ap.add_argument('--batch', type=int, default=128)
    ap.add_argument('--lr', type=float, default=1e-3)
    ap.add_argument('--lambda_reg', type=float, default=1e-3)
    ap.add_argument('--device', type=str,
        default='cuda' if torch.cuda.is_available() else 'cpu')
    args = ap.parse_args()

    set_seed(args.seed)
    device = torch.device(args.device)
    tfm = transforms.ToTensor()

    trainA = SplitMNIST('./data', train=True, digits={0,1,2,3,4}, transform=tfm)
    trainB = SplitMNIST('./data', train=True, digits={5,6,7,8,9}, transform=tfm)
    testA_full = datasets.MNIST('./data', train=False, transform=tfm, download=True)
    idxA = [i for i, (_, y) in enumerate(testA_full) if y in {0,1,2,3,4}]
    testA = Subset(testA_full, idxA)
    testB_full = datasets.MNIST('./data', train=False, transform=tfm, download=True)
    idxB = [i for i, (_, y) in enumerate(testB_full) if y in {5,6,7,8,9}]
    testB = Subset(testB_full, idxB)

    from torch.utils.data import DataLoader
    loaderA = DataLoader(trainA, batch_size=args.batch, shuffle=True, num_workers=2)
    loaderB = DataLoader(trainB, batch_size=args.batch, shuffle=True, num_workers=2)
    loaderTestA = DataLoader(testA, batch_size=256, shuffle=False, num_workers=2)
    loaderTestB = DataLoader(testB, batch_size=256, shuffle=False, num_workers=2)

    model = MLP().to(device)
    print('Training Task A...')
    theta_A = train_task_A(model, loaderA, device, epochs=args.epochsA, lr=args.lr)
    lossA0, accA0 = evaluate(model, loaderTestA, device)
    print(f'After A:  lossA={lossA0:.4f} accA={accA0:.4f}')

    print('Fine-tuning on Task B with fixed lambda (drift penalty)...')
    finetune_task_B_with_drift(model, loaderB, device, theta_A,
                               lam=args.lambda_reg, epochs=args.epochsB, lr=args.lr)
    lossA1, accA1 = evaluate(model, loaderTestA, device)
    lossB1, accB1 = evaluate(model, loaderTestB, device)
    print(f'After B:  accA={accA1:.4f}, accB={accB1:.4f}, forgetting={accA0-accA1:.4f}')

if __name__ == '__main__':
    main()
\end{lstlisting}
